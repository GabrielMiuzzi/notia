cmake_minimum_required(VERSION 3.22)
project(notia_qwen3_tts_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
endif()
set(QWEN3_TTS_COREML OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${QWEN3_TTS_SOURCE}" qwen3-tts)
add_library(qwen3_tts_runtime SHARED bridge.cpp)
target_compile_definitions(qwen3_tts_runtime PRIVATE QWEN3_TTS_STATIC)
target_include_directories(qwen3_tts_runtime PRIVATE "${QWEN3_TTS_SOURCE}/src")
target_link_libraries(qwen3_tts_runtime PRIVATE qwen3_tts)
